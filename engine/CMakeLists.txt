FILE (GLOB_RECURSE SRC "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

ADD_EXECUTABLE (engine ${SRC})

TARGET_COMPILE_DEFINITIONS (engine PUBLIC "_CRT_SECURE_NO_WARNINGS")

TARGET_INCLUDE_DIRECTORIES (engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")

IF (CMAKE_BUILD_TYPE STREQUAL "Debug")
	TARGET_COMPILE_DEFINITIONS (engine PRIVATE _ITERATOR_DEBUG_LEVEL=2)
ELSEIF (CMAKE_BUILD_TYPE STREQUAL "Release")
	TARGET_COMPILE_DEFINITIONS (engine PRIVATE _ITERATOR_DEBUG_LEVEL=0)
ENDIF ()

TARGET_LINK_LIBRARIES (engine PUBLIC Vulkan::Vulkan)
TARGET_LINK_LIBRARIES (engine PUBLIC cgltf)
TARGET_LINK_LIBRARIES (engine PUBLIC clang)
TARGET_LINK_LIBRARIES (engine PUBLIC flecs)
TARGET_LINK_LIBRARIES (engine PUBLIC fs)
TARGET_LINK_LIBRARIES (engine PUBLIC glslang)
TARGET_LINK_LIBRARIES (engine PUBLIC imgui)
TARGET_LINK_LIBRARIES (engine PUBLIC spirv_reflect)
TARGET_LINK_LIBRARIES (engine PUBLIC stb)

TARGET_PRECOMPILE_HEADERS (engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include/ti_pch.h")
